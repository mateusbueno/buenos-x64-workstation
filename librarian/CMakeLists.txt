# Story 5.5: the keyword table (librarian/data/category_keywords.json) is also compiled into the library as a
# fallback, so the binary needs no data file. Regenerated at configure time whenever the JSON changes.
set(VELORA_CATEGORY_KEYWORDS_JSON "${CMAKE_CURRENT_SOURCE_DIR}/data/category_keywords.json")
set(VELORA_CATEGORY_KEYWORDS_CPP "${CMAKE_CURRENT_BINARY_DIR}/category_keywords_embedded.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${VELORA_CATEGORY_KEYWORDS_JSON}")
file(READ "${VELORA_CATEGORY_KEYWORDS_JSON}" VELORA_CATEGORY_KEYWORDS_TEXT)
file(WRITE "${VELORA_CATEGORY_KEYWORDS_CPP}.tmp"
"// Generated by librarian/CMakeLists.txt from data/category_keywords.json. Do not edit.
#include \"velora/librarian/category_suggest.hpp\"
namespace velora::librarian {
std::string_view KeywordTable::embedded_json() {
  static constexpr char kJson[] = R\"VLRJSON(${VELORA_CATEGORY_KEYWORDS_TEXT})VLRJSON\";
  return std::string_view(kJson, sizeof(kJson) - 1);
}
}  // namespace velora::librarian
")
execute_process(COMMAND ${CMAKE_COMMAND} -E copy_if_different "${VELORA_CATEGORY_KEYWORDS_CPP}.tmp" "${VELORA_CATEGORY_KEYWORDS_CPP}")

add_library(velora_librarian STATIC src/import.cpp src/batch.cpp src/category_suggest.cpp "${VELORA_CATEGORY_KEYWORDS_CPP}")
add_library(velora::librarian ALIAS velora_librarian)
target_include_directories(velora_librarian PUBLIC include)
target_link_libraries(velora_librarian PUBLIC velora_format velora_sfz velora_host)
target_compile_options(velora_librarian PRIVATE
  $<$<OR:$<CXX_COMPILER_ID:GNU>,$<CXX_COMPILER_ID:Clang>,$<CXX_COMPILER_ID:AppleClang>>:-Wall -Wextra>)

add_executable(librarian src/main.cpp)
target_link_libraries(librarian PRIVATE velora_librarian)
